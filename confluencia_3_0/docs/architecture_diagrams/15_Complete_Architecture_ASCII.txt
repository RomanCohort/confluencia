         │   ├── heterogeneity.step() → 更新 clone_frequencies

         │   ├── fibroblast.step() → 更新 caf_activation, ecm_density

         │
         ├── TreatmentManager.step()

         │   ├── immunotherapy.step() → 更新 checkpoint_inhibition
         │   ├── targeted.step() → 更新 pathway_inhibition
         │   ├── radiotherapy.step() → 更新 radiation_damage
         │   └── circrna_therapy.step() → 更新 circrna_effect
         │
         ├── CircRNAManager.step()
         │   ├── assess_immunogenicity() ──→ immune_sensing.py ──→ Backend 三层降级
         │   │   └── esm2 ──→ vienna ──→ heuristic
         │   ├── predict_structure() ──→ structure_prediction.py ──→ ViennaRNA MFE

         │   │   ├── extract_signals() → tpe → equivariant_backbone → CircPairformer → IRS Pair → Structure Head
